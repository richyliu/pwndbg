   265         1       7000.0   7000.0      0.0      def eX(size, address, data, hex=True) -> None:
   266                                                   """
   267                                                   This relies on windbg's default hex encoding being enforced
   268                                                   """
   269                                                   if not data:
   270                                                       print("Cannot write empty data into memory.")
   271                                                       return
   272                                           
   273                                                   if hex:
   274                                                       # Early validation if all data is hex
   275                                                       for string in data:
   276                                                           if string.startswith("0x"):
   277                                                               string = string[2:]
   278                                           
   279                                                           if any(ch not in "0123456789abcdefABCDEF" for ch in string):
   280                                                               print(
   281                                                                   "Incorrect data format: it must all be a hex value (0x1234 or 1234, both interpreted as 0x1234)"
   282                                                               )
   283                                                               return
   284                                           
   285                                                   writes = 0
   286                                                   for i, string in enumerate(data):
   287                                                       if hex:
   288                                                           if string.startswith("0x"):
   289                                                               string = string[2:]
   290                                           
   291                                                           string = string.rjust(size * 2, "0")
   292                                           
   293                                                           data = codecs.decode(string, "hex")
   294                                                       else:
   295                                                           data = string
   296                                           
   297                                                       if pwndbg.aglib.arch.endian == "little":
   298                                                           data = data[::-1]
   299                                           
   300                                                       try:
   301                                                           pwndbg.aglib.memory.write(address + (i * size), data)
   302                                                           writes += 1
   303                                                       except pwndbg.dbg_mod.Error:
   304                                                           print("Cannot access memory at address %#x" % address)
   305                                                           if writes > 0:
   306                                                               print("(Made %d writes to memory; skipping further writes)" % writes)
   307                                                           return

   987                                                       bucket_count = 8  # taken from src/internal/abi/map.go commit 1b4f1dc
   988                                                       word = word_size()
   989                                                       offsets = self.field_offsets()
   990                                                       val = pwndbg.aglib.memory.read(addr, offsets["$size"])
   991                                                       load = lambda off, sz: load_uint(val[off : off + sz])
   992                                                       num_buckets = 1 << load(offsets["B"], 1)
   993                                                       num_oldbuckets = num_buckets >> 1
   994                                                       oldbucket_base = load(offsets["oldbuckets"], word)
   995                                                       bucket_base = load(offsets["buckets"], word)
   996                                                       keysize = self.key.size()
   997                                                       valsize = self.val.size()
   998                                                       # technically need to worry about padding but every go arch has max alignment of 8 and bucket count is 8
   999                                                       # so padding is never actually possible
  1000                                                       [tophash_start, keys_start, vals_start, overflow_start, bucket_size] = compute_offsets(
  1001                                                           [
  1002                                                               (bucket_count, 1),
  1003                                                               (keysize * bucket_count, 1),
  1004                                                               (valsize * bucket_count, 1),
  1005                                                               (word, word),
  1006                                                           ]
  1007                                                       )
  1008                                                       ret = []
  1009                                                       for i in range(num_buckets):
  1010                                                           bucket_ptr = bucket_base + bucket_size * i
  1011                                                           if oldbucket_base and i < num_oldbuckets:
  1012                                                               oldbucket_ptr = oldbucket_base + bucket_size * i
  1013                                                               oldbucket = pwndbg.aglib.memory.read(oldbucket_ptr, bucket_size)
  1014                                                               if not (1 < oldbucket[tophash_start] < 5):  # !evacuated(bucket)
  1015                                                                   bucket_ptr = oldbucket_ptr

  1017                                                               bucket = pwndbg.aglib.memory.read(bucket_ptr, bucket_size)
  1018                                                               for j in range(bucket_count):

  1020                                                                       key_ptr = bucket_ptr + keys_start + j * keysize
  1021                                                                       val_ptr = bucket_ptr + vals_start + j * valsize
  1022                                                                       k = self.key.dump(key_ptr, fmt)
  1023                                                                       v = self.val.dump(val_ptr, fmt)
  1024                                                                       ret.append((key_ptr, val_ptr, k, v))
  1025                                                               bucket_ptr = load_uint(bucket[overflow_start : overflow_start + word])
  1026                                                       # sort map by key, using integer comparison if possible
  1027                                                       try:
  1028                                                           ret.sort(key=lambda t: int(t[2], 0))
  1029                                                       except ValueError:
  1030                                                           ret.sort(key=lambda t: t[2])
  1031                                                       formatted = []
  1032                                                       for kp, vp, k, v in ret:
  1033                                                           prefix = fmt.fmt_debug(f"(key @ {kp:#x}, val @ {vp:#x}) ")
  1034                                                           formatted.append(f"{prefix}{k}: {v}")
